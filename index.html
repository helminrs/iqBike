function saveDriverWaNumber() { const waVal = document.getElementById('driverWaInput').value.trim(); // Validasi ketat format nomor WhatsApp Indonesia standar (minimal 10 digit, maksimal 13 digit, harus berawalan 62 atau 08) let cleanedWa = waVal.replace(/\D/g, ''); if (cleanedWa.startsWith('08')) { cleanedWa = '62' + cleanedWa.substring(1); } // Cek apakah format valid (harus berawalan 628 dan panjang total 10 sampai 13 digit) const waRegex = /^628\d{7,10}$/; if (!waRegex.test(cleanedWa)) { alert("❌ NOMOR WHATSAPP TIDAK VALID!\n\nNomor WhatsApp Driver harus nomor seluler Indonesia yang sah (Contoh: 628123456789 atau 628123456789, minimal 10 digit dan berawalan 08/628). Nomor asal/palsu ditolak sistem."); return; } const newWa = cleanedWa; // Jika nomor WA diganti, hapus dokumen lama di Firestore agar ID tidak nyangkut/duplikat if (newWa !== currentDriverWa) { db.collection("iqbike_drivers").doc(currentDriverWa).delete().catch(() => {}); currentDriverWa = newWa; } localStorage.setItem('iqbike_driver_wa', currentDriverWa); document.getElementById('driverWaInput').value = currentDriverWa; // Langsung push status ke cloud dengan ID baru yang terverifikasi valid syncDriverToCloud(); alert(`✅ ID Nomor WhatsApp Driver (${currentDriverWa}) Berhasil Disimpan & Terkunci ke Sistem!`); }
